cmake_minimum_required(VERSION 3.20)
project(GPUBosciaConsumer LANGUAGES CXX)

if(GPUBOSCIA_SOURCE_ROOT)
    add_subdirectory("${GPUBOSCIA_SOURCE_ROOT}" gpuboscia-dependency)
    if(TARGET test_cpu OR TARGET gpuboscia_cli)
        message(FATAL_ERROR "Adding GPUBoscia as a dependency must not build its tests/examples by default")
    endif()
else()
    if(NOT GPUBOSCIA_CONSUMER_COMPONENT)
        set(GPUBOSCIA_CONSUMER_COMPONENT Core)
    endif()
    find_package(GPUBoscia 0.1 CONFIG REQUIRED COMPONENTS ${GPUBOSCIA_CONSUMER_COMPONENT})
endif()
add_executable(consumer main.cpp second.cpp)
target_link_libraries(consumer PRIVATE gpuboscia::core)
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
